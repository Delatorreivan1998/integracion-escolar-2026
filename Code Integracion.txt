/**********************

function buildMenu_() {
  const ui = SpreadsheetApp.getUi();

  ui.createAddonMenu()
    .addItem('🧩 INTEGRACIONES PADRES (fila seleccionada)', 'sendMailForSelectedRow')
    .addItem('🔁 Forzar reenvío (fila seleccionada)', 'forceResendForSelectedRow')
    .addSeparator()
    .addItem('📤 Enviar pendientes (sin fecha en G)', 'sendPending')
    .addSeparator()

  ui.createMenu('📧 Mails Aut. Padres 2026')
    .addItem('🧩 INTEGRACIONES PADRES (fila seleccionada)', 'sendMailForSelectedRow')

    .addItem('📤 Enviar pendientes', 'sendPending')
    .addToUi();

function onOpen(e) { buildMenu_(); }

function mostrarMenuAhora() {

}


/**********************
 * ACCIONES
 **********************/
function sendMailForSelectedRow() {
  sendForActiveRow_(false);
}

function forceResendForSelectedRow() {
  sendForActiveRow_(true);
}

/**
 * Usa la fila seleccionada (row del rango activo),
 * pero siempre toma datos desde la hoja "Casos Aprobados"
 * aunque estés parado en otra pestaña.
 */
function sendForActiveRow_(force) {
  const ss = SpreadsheetApp.getActive();
  const targetSheet = ss.getSheetByName(SHEET_NAME);
  if (!targetSheet) {
    SpreadsheetApp.getUi().alert('No existe la hoja "' + SHEET_NAME + '".');
    return;
  }

  const rng = ss.getActiveRange();
  if (!rng) {
    SpreadsheetApp.getUi().alert('Seleccioná una celda en la fila que querés enviar.');
    return;
  }

  const row = rng.getRow();
  if (row < 2) return;

  sendForRow_(targetSheet, row, force);
}

/**
 * Pendientes: SOLO envía si:
 * - G está vacío
 * - F es un mail real (ej. pepito@gmail.com)
 * - y NO es "E-MAIL" / "E- mail" / "E mail" etc.
 *
 * Importante: si F no es mail real, NO envía y NO pinta nada.
 */
function sendPending() {
  const ss = SpreadsheetApp.getActive();
  const sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) throw new Error('No existe la hoja: ' + SHEET_NAME);

  const lastRow = sh.getLastRow();
  if (lastRow < 2) return;

  for (let row = 2; row <= lastRow; row++) {
    const alreadySent = sh.getRange(row, SENT_COL).getValue();
    if (alreadySent) continue; // ya tiene fecha en G

    const rawEmail = String(sh.getRange(row, EMAIL_COL).getValue() || '').trim();

    // Saltea placeholders tipo "E-MAIL" y cualquier cosa que no sea un email real
    if (!isRealEmail_(rawEmail)) continue;

    sendForRow_(sh, row, false);
    Utilities.sleep(600);
  }
}


/**********************
 * PDF
 **********************/
function getPdfBlob_() {
  const folder = DriveApp.getFolderById(FOLDER_ID);

  let files = folder.getFilesByName(PDF_NAME);
  if (files.hasNext()) return files.next().getBlob().setName(PDF_NAME);

  // fallback: normaliza espacios para evitar fallos por doble/tri espacios
  files = folder.getFiles();
  const target = PDF_NAME.replace(/\s+/g, ' ').trim();
  while (files.hasNext()) {
    const f = files.next();
    const nameNorm = f.getName().replace(/\s+/g, ' ').trim();
    if (nameNorm === target) return f.getBlob().setName(f.getName());
  }

  throw new Error('No se encontró el PDF "' + PDF_NAME + '" en la carpeta indicada.');
}


/**********************
 * ENVÍO (pinta SOLO celda Mail en F)
 **********************/
function sendForRow_(sh, row, force) {
  const emailCell = sh.getRange(row, EMAIL_COL);
  const rawEmail = String(emailCell.getValue() || '').trim();

  const sentCell = sh.getRange(row, SENT_COL);
  const alreadySent = sentCell.getValue();

  // Si no es mail real (incluye "E-MAIL"), marcar inválido y no enviar
  if (!isRealEmail_(rawEmail)) {
    sentCell.setValue('Sin email válido');
    emailCell.setBackground(COLOR_INVALID);
    return;
  }

  if (!force && alreadySent) return;

  // Amarillo (queda amarillo como pediste)
  emailCell.setBackground(COLOR_SENDING);

  const pdfBlob = getPdfBlob_();

  GmailApp.sendEmail(rawEmail, SUBJECT, BODY_TEXT, {
    htmlBody: BODY_HTML,
    attachments: [pdfBlob],
    name: SENDER_NAME
  });

  // Fecha en G
  sentCell.setValue(new Date());
}


/**********************
 * VALIDACIÓN: MAIL REAL (bloquea "E-MAIL")
 **********************/
function isRealEmail_(value) {
  if (!value) return false;

  const v = String(value).trim();
  if (!v) return false;

  // Normaliza para detectar placeholders: "E-MAIL", "E mail", "E_mail", etc.
  const normalized = v.toLowerCase().replace(/\s+/g, '').replace(/[\-_.]/g, '');
  if (normalized === 'email') return false;

  // Validación práctica de email real
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v);
}


/**********************
 * TRIGGER INSTALABLE onOpen
 **********************/
function installOnOpenTrigger() {
  ScriptApp.getProjectTriggers()
    .filter(t => t.getHandlerFunction() === 'onOpen')
    .forEach(t => ScriptApp.deleteTrigger(t));

  ScriptApp.newTrigger('onOpen')
    .forSpreadsheet(SpreadsheetApp.getActive())
    .onOpen()
    .create();

  SpreadsheetApp.getUi().alert('🧷 Trigger instalable onOpen creado.');
}
